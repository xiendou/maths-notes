\subsection{Natural numbers}
For this lecture, the natural numbers start at $1$:
\[
\mathbb{N}=\{1,2,3,4,\ldots\}.
\]
Thus $0\notin\mathbb{N}$.

\subsection{Operations and order on $\mathbb{N}$}
The natural numbers are closed under addition and multiplication:
\[
m,n\in\mathbb{N}\quad\Longrightarrow\quad
m+n\in\mathbb{N}\quad\text{and}\quad mn\in\mathbb{N}.
\]
For any $m,n\in\mathbb{N}$, exactly one of $m>n$, $m=n$, or $m<n$
holds. Also, $m\geq n$ means $m>n$ or $m=n$.

\subsection{Properties}
For $m,n,p\in\mathbb{N}$:
\begin{align*}
m+n &= n+m, & mn &= nm && \text{(commutativity)},\\
(m+n)+p &= m+(n+p), & (mn)p &= m(np) && \text{(associativity)},\\
m(n+p) &= mn+mp &&&& \text{(distributivity)}.
\end{align*}
Order is transitive:
\[
m>n\ \text{and}\ n>p\quad\Longrightarrow\quad m>p,
\]
and similarly
\[
m\geq n\ \text{and}\ n\geq p\quad\Longrightarrow\quad m\geq p.
\]

\subsection{Why introduce integers?}
Some equations have no solution in $\mathbb{N}$. For example,
$x+1=0$ requires $x=-1$. This motivates the integers:
\[
\mathbb{Z}=\{\ldots,-3,-2,-1,0,1,2,3,\ldots\}.
\]
